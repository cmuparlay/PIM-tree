function(add_host_executable NAME_PREFIX SOURCE TYPE)
    set(BIN_NAME "${NAME_PREFIX}_${TYPE}")

    message("SOURCE = ${SOURCE}")
    message("TYPE = ${TYPE}")
    message("HOST_NAME = ${BIN_NAME}")

    add_executable(${BIN_NAME} ${SOURCE})

    if(NOT(TYPE STREQUAL "no_shadow_subtree"))
        target_compile_definitions(${BIN_NAME} PUBLIC SHADOW_SUBTREE=1)
    endif()

    if (TYPE STREQUAL "energy")
        target_compile_definitions(${BIN_NAME} PUBLIC DPU_ENERGY=1)
    endif()

    target_link_libraries(${BIN_NAME} PUBLIC Threads::Threads)
    target_link_libraries(${BIN_NAME} PUBLIC PAPI::PAPI)
    target_link_libraries(${BIN_NAME} PUBLIC stdc++fs)
    target_link_libraries(${BIN_NAME} PUBLIC dpu)

    target_compile_definitions(${BIN_NAME} PUBLIC NR_DPUS=${NR_DPUS})
    target_compile_options(${BIN_NAME} PUBLIC -Wall -Wextra -g -O3)

    target_include_directories(${BIN_NAME} PUBLIC "/usr/include/dpu")
    target_include_directories(${BIN_NAME} PUBLIC "/usr/include/dpu")

    # target_include_directories(${NAME} SYSTEM PUBLIC ${INCLUDE_DIRS})
    # target_include_directories(${NAME} PUBLIC ${SRC_INCLUDE_DIRS})
    # # target_compile_definitions(${NAME} PUBLIC PARLAY_OPENMP)
    # target_link_libraries(${NAME} PUBLIC OpenMP::OpenMP_CXX)
    # target_link_libraries(${NAME} PUBLIC absl::flat_hash_map)
    # target_link_libraries(${NAME} PUBLIC absl::container_memory)
    # target_link_libraries(${NAME} PUBLIC absl::core_headers)
    # target_link_libraries(${NAME} PUBLIC absl::hash_container_defaults)
    # target_link_libraries(${NAME} PUBLIC absl::raw_hash_map)
    # target_link_libraries(${NAME} PUBLIC absl::algorithm_container)
    # target_link_libraries(${NAME} PUBLIC absl::type_traits)
    # target_link_libraries(${NAME} PUBLIC absl::hash)

    # if(CMAKE_BUILD_TYPE MATCHES Debug)
    #     target_compile_definitions(${NAME} PUBLIC DEBUG)
    # elseif(CMAKE_BUILD_TYPE MATCHES Release)
    # else()
    # endif()
endfunction()
